# Always exercise both modes, regardless of the mode selected by the consumer.
if(SIMPLE_CPP_LOGGER_COMPILED)
  set(LOGGER_COMPILED_TEST_TARGET SimpleCppLogger)
else()
  add_library(SimpleCppLoggerTestLibrary STATIC ../src/Logger.cpp)
  target_include_directories(SimpleCppLoggerTestLibrary PUBLIC ../include)
  target_compile_features(SimpleCppLoggerTestLibrary PUBLIC cxx_std_17)
  target_compile_definitions(SimpleCppLoggerTestLibrary PUBLIC SIMPLE_CPP_LOGGER_COMPILED)
  set(LOGGER_COMPILED_TEST_TARGET SimpleCppLoggerTestLibrary)
endif()

foreach(mode IN ITEMS HeaderOnly Compiled)
  foreach(test IN ITEMS Compatibility TranslationUnits)
    add_executable(Logger${test}${mode} ${test}.cpp)
    if(test STREQUAL "TranslationUnits")
      target_sources(Logger${test}${mode} PRIVATE TranslationUnitPeer.cpp)
    endif()
    if(mode STREQUAL "Compiled")
      target_link_libraries(Logger${test}${mode} PRIVATE ${LOGGER_COMPILED_TEST_TARGET})
    else()
      target_include_directories(Logger${test}${mode} PRIVATE ../include)
      target_compile_features(Logger${test}${mode} PRIVATE cxx_std_17)
    endif()
    add_test(NAME Logger${test}${mode} COMMAND Logger${test}${mode})
  endforeach()
endforeach()
